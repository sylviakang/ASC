2把 choose 拆掉

第 29 堂的 choose 一次做完「判斷 + 換題」。現在答對要先等 2 秒，所以拆成兩段：

  1. 停留滿 3 秒：只判斷對錯、設定 marked 和 phase。
  2. phase 改變後：WRONG 等 5 秒、CORRECT 等 2 秒再換題或結束。

3when 處理三種狀態

選完之後（最終版）
LaunchedEffect(phase, index) {
    when (phase) {
        Phase.WRONG -> {
            val start = System.currentTimeMillis()
            while (true) {
                val passed = System.currentTimeMillis() - start
                lock = 1f - dwellProgress(passed, 5000)
                if (passed >= 5000) break
                delay(16)
            }
            marked = -1
            phase = Phase.PLAYING
        }
        Phase.CORRECT -> {
            delay(2000)                          // 綠框顯示 2 秒
            marked = -1
            if (index == questions.lastIndex) {
                onFinish(
                    GameResult(
                        correct = questions.size - wrongIds.size,
                        wrong = wrongIds.size,
                        durationSec = (elapsedMs / 1000).toInt(),
                        endMillis = System.currentTimeMillis()
                    )
                )
            } else {
                colors = nextColorIndexes(OptionPalette.size, colors)
                index++
                phase = Phase.PLAYING
            }
        }
        Phase.PLAYING -> {}
    }
}
  • when (phase) 像 C# 的 switch，把三種狀態都列出來，一眼看懂。
  • CORRECT：delay(2000) 讓綠框顯示 2 秒 → 清 marked → 最後一題就 onFinish，否則換色、index++、回到 PLAYING。
  • Phase.PLAYING -> {}：什麼都不做。when 當成敘述用時可以省略，但寫出來比較完整。
補充
綠框怎麼出現的？上一堂的 borderColor 已經寫好：phase 不是 WRONG 就是綠色。marked = hovered 之後就自動出現，不用改畫面程式。

4換題時 index 改變會發生什麼

index++ 後，兩個 LaunchedEffect 的 key 都包含 index，所以都會重新開始；停留計時 dwell 歸零，從新的一題重新計算。游標位置不變（沒有自動校正），這符合題本。

5完整的 QuizScreen

到這裡答題頁已經是最終版本，對照下面的完整程式檢查自己的檔案：

QuizScreen.kt（最終版）
enum class Phase { PLAYING, WRONG, CORRECT }

val OptionPalette = listOf(
    Color(0xFFEF9A9A), Color(0xFF90CAF9), Color(0xFFA5D6A7), Color(0xFFFFE082),
    Color(0xFFCE93D8), Color(0xFF80CBC4), Color(0xFFFFAB91), Color(0xFFB0BEC5)
)

const val PX_PER_RAD = 800f      // 傾斜 1 弧度（約 57 度）游標移動 800 px
const val CURSOR_RADIUS = 36f    // 游標半徑 px

@Composable
fun QuizScreen(
    allQuestions: List<Question>,
    onExit: () -> Unit,
    onFinish: (GameResult) -> Unit
) {
    val context = LocalContext.current
    val questions = remember { allQuestions.shuffled().take(10) }
    var index by remember { mutableIntStateOf(0) }
    val q = questions[index]
    var colors by remember { mutableStateOf(nextColorIndexes(OptionPalette.size, null)) }
    val wrongIds = remember { mutableSetOf<String>() }
    var showExit by remember { mutableStateOf(false) }

    // ---------- 計時：進背景暫停 ----------
    var elapsedMs by remember { mutableLongStateOf(0L) }
    var resumed by remember { mutableStateOf(true) }
    val lifecycleOwner = LocalLifecycleOwner.current
    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            if (event == Lifecycle.Event.ON_RESUME) resumed = true
            if (event == Lifecycle.Event.ON_PAUSE) resumed = false
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
    }
    LaunchedEffect(resumed) {
        if (!resumed) return@LaunchedEffect
        var last = System.currentTimeMillis()
        while (true) {
            delay(100)
            val now = System.currentTimeMillis()
            elapsedMs += now - last
            last = now
        }
    }

    // ---------- 陀螺儀：角速度積分成角度 ----------
    var angleX by remember { mutableFloatStateOf(0f) }
    var angleY by remember { mutableFloatStateOf(0f) }
    var baseX by remember { mutableFloatStateOf(0f) }
    var baseY by remember { mutableFloatStateOf(0f) }
    DisposableEffect(Unit) {
        val sm = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager
        val gyro = sm.getDefaultSensor(Sensor.TYPE_GYROSCOPE)
        var lastNs = 0L
        val listener = object : SensorEventListener {
            override fun onSensorChanged(e: SensorEvent) {
                if (lastNs != 0L) {
                    val dt = (e.timestamp - lastNs) / 1_000_000_000f
                    angleX = integrate(angleX, e.values[0], dt)
                    angleY = integrate(angleY, e.values[1], dt)
                }
                lastNs = e.timestamp
            }
            override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {}
        }
        sm.registerListener(listener, gyro, SensorManager.SENSOR_DELAY_GAME)
        onDispose { sm.unregisterListener(listener) }
    }

    // ---------- 選項位置與游標 ----------
    val rects = remember { mutableStateListOf<FloatArray?>(null, null, null, null) }
    var originX by remember { mutableFloatStateOf(0f) }
    var originY by remember { mutableFloatStateOf(0f) }
    val ready = rects.all { it != null }
    val left = if (ready) rects.minOf { it!![0] } else 0f
    val top = if (ready) rects.minOf { it!![1] } else 0f
    val right = if (ready) rects.maxOf { it!![2] } else 0f
    val bottom = if (ready) rects.maxOf { it!![3] } else 0f
    val cursorX = clamp(
        cursorPos((left + right) / 2, angleY, baseY, PX_PER_RAD),
        left + CURSOR_RADIUS, right - CURSOR_RADIUS
    )
    val cursorY = clamp(
        cursorPos((top + bottom) / 2, angleX, baseX, PX_PER_RAD),
        top + CURSOR_RADIUS, bottom - CURSOR_RADIUS
    )
    val hovered = if (ready) hitIndex(cursorX, cursorY, rects.map { it!! }) else -1

    // ---------- 停留選答與答對答錯狀態 ----------
    var phase by remember { mutableStateOf(Phase.PLAYING) }
    var marked by remember { mutableIntStateOf(-1) }       // 顯示紅框或綠框的選項
    var dwell by remember { mutableFloatStateOf(0f) }      // 停留進度 0～1
    var lock by remember { mutableFloatStateOf(0f) }       // 鎖定剩餘 1～0

    LaunchedEffect(hovered, phase, index) {
        dwell = 0f
        if (phase != Phase.PLAYING || hovered < 0) return@LaunchedEffect
        val start = System.currentTimeMillis()
        while (dwell < 1f) {
            delay(16)
            dwell = dwellProgress(System.currentTimeMillis() - start)
        }
        marked = hovered
        if (hovered == q.answerIndex) {
            phase = Phase.CORRECT
        } else {
            wrongIds.add(q.qid)
            Toast.makeText(context, "選擇錯誤", Toast.LENGTH_SHORT).show()
            phase = Phase.WRONG
        }
    }

    LaunchedEffect(phase, index) {
        when (phase) {
            Phase.WRONG -> {
                val start = System.currentTimeMillis()
                while (true) {
                    val passed = System.currentTimeMillis() - start
                    lock = 1f - dwellProgress(passed, 5000)
                    if (passed >= 5000) break
                    delay(16)
                }
                marked = -1
                phase = Phase.PLAYING
            }
            Phase.CORRECT -> {
                delay(2000)
                marked = -1
                if (index == questions.lastIndex) {
                    onFinish(
                        GameResult(
                            correct = questions.size - wrongIds.size,
                            wrong = wrongIds.size,
                            durationSec = (elapsedMs / 1000).toInt(),
                            endMillis = System.currentTimeMillis()
                        )
                    )
                } else {
                    colors = nextColorIndexes(OptionPalette.size, colors)
                    index++
                    phase = Phase.PLAYING
                }
            }
            Phase.PLAYING -> {}
        }
    }

    BackHandler { showExit = true }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .onGloballyPositioned {
                originX = it.positionInRoot().x
                originY = it.positionInRoot().y
            }
            .pointerInput(Unit) {
                detectTapGestures(onDoubleTap = {
                    // 雙擊校正：以現在的角度為新的基準，游標回到中心
                    baseX = angleX
                    baseY = angleY
                })
            }
    ) {
        Column(Modifier.fillMaxSize()) {
            Row(
                modifier = Modifier.fillMaxWidth().padding(4.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                TextButton(onClick = { showExit = true }) { Text("‹ 返回", fontSize = 18.sp) }
                Text("${index + 1}/${questions.size}", fontSize = 20.sp)
                Spacer(Modifier.weight(1f))
                Text(formatClock((elapsedMs / 1000).toInt()), fontSize = 20.sp)
                Spacer(Modifier.padding(8.dp))
            }
            LinearProgressIndicator(
                progress = { (index + 1) / questions.size.toFloat() },
                modifier = Modifier.fillMaxWidth().padding(horizontal = 16.dp)
            )
            Text(
                q.question,
                fontSize = 22.sp,
                modifier = Modifier.fillMaxWidth().padding(24.dp),
                textAlign = TextAlign.Center
            )
            Column(
                modifier = Modifier.fillMaxWidth().weight(1f).padding(16.dp),
                verticalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                for (row in 0..1) {
                    Row(Modifier.weight(1f), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                        for (col in 0..1) {
                            val i = row * 2 + col
                            val borderColor = if (phase == Phase.WRONG) Color.Red else Color(0xFF2E7D32)
                            Box(
                                modifier = Modifier
                                    .weight(1f)
                                    .fillMaxSize()
                                    .background(OptionPalette[colors[i]], RoundedCornerShape(16.dp))
                                    .then(
                                        if (marked == i) Modifier.border(6.dp, borderColor, RoundedCornerShape(16.dp))
                                        else Modifier
                                    )
                                    .onGloballyPositioned {
                                        val b = it.boundsInRoot()
                                        val r = floatArrayOf(b.left, b.top, b.right, b.bottom)
                                        if (rects[i]?.contentEquals(r) != true) rects[i] = r
                                    }
                                    .padding(12.dp),
                                contentAlignment = Alignment.Center
                            ) {
                                Text(q.options[i], fontSize = 20.sp, textAlign = TextAlign.Center)
                            }
                        }
                    }
                }
            }
        }

        // 游標與圓形進度條畫在最上層
        Canvas(Modifier.fillMaxSize()) {
            if (!ready) return@Canvas
            val c = Offset(cursorX - originX, cursorY - originY)
            drawCircle(Color(0xFF37474F), radius = CURSOR_RADIUS * 0.6f, center = c)
            val ringSize = Size(CURSOR_RADIUS * 2, CURSOR_RADIUS * 2)
            val ringTopLeft = Offset(c.x - CURSOR_RADIUS, c.y - CURSOR_RADIUS)
            if (phase == Phase.WRONG) {
                drawArc(Color.Red, -90f, 360f * lock, false, ringTopLeft, ringSize, style = Stroke(8f))
            } else if (dwell > 0f) {
                drawArc(Color(0xFF1565C0), -90f, 360f * dwell, false, ringTopLeft, ringSize, style = Stroke(8f))
            }
        }
    }

    if (showExit) {
        AlertDialog(
            onDismissRequest = { showExit = false },
            title = { Text("確認退出") },
            text = { Text("確定要退出答題嗎？本局不會保存。") },
            confirmButton = { TextButton(onClick = onExit) { Text("退出") } },
            dismissButton = { TextButton(onClick = { showExit = false }) { Text("取消") } }
        )
    }
}
